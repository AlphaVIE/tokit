// HTTP/1.1 over std::net. The reference interpreter includes this file and
// native programs embed it, so both backends share one implementation.
#[derive(Clone)]
pub(crate) struct __TokHttpRequest { pub(crate) method: String, pub(crate) path: String, pub(crate) query: String, pub(crate) headers: Vec<(String, String)>, pub(crate) body: String }
#[derive(Clone)]
pub(crate) struct __TokHttpResponse { pub(crate) status: i32, pub(crate) headers: Vec<(String, String)>, pub(crate) body: String }

const __TOK_HTTP_HEAD_LIMIT: usize = 64 * 1024;
const __TOK_HTTP_BODY_LIMIT: usize = 16 * 1024 * 1024;

/// Whether a `--allow-net` grant covers `addr`: an exact `host:port`, or `*`.
pub(crate) fn __tok_http_allowed(grant: Option<&str>, addr: &str) -> bool {
    grant.is_some_and(|grant| grant == "*" || grant == addr)
}

pub(crate) fn __tok_http_reason(status: i32) -> &'static str {
    match status {
        200 => "OK", 201 => "Created", 202 => "Accepted", 204 => "No Content",
        301 => "Moved Permanently", 302 => "Found", 304 => "Not Modified",
        400 => "Bad Request", 401 => "Unauthorized", 403 => "Forbidden", 404 => "Not Found",
        405 => "Method Not Allowed", 409 => "Conflict", 413 => "Content Too Large",
        422 => "Unprocessable Content", 500 => "Internal Server Error", 503 => "Service Unavailable",
        _ => "Status",
    }
}

/// Header names are case-insensitive, so they are lowercased; repeated
/// headers are joined with ", " as HTTP permits.
pub(crate) fn __tok_http_add_header(headers: &mut Vec<(String, String)>, name: &str, value: &str) {
    let name = name.trim().to_ascii_lowercase();
    let value = value.trim().to_owned();
    match headers.iter_mut().find(|(existing, _)| *existing == name) {
        Some((_, existing)) => { existing.push_str(", "); existing.push_str(&value); }
        None => headers.push((name, value)),
    }
}

pub(crate) fn __tok_http_header<'a>(headers: &'a [(String, String)], name: &str) -> Option<&'a str> {
    headers.iter().find(|(existing, _)| existing == name).map(|(_, value)| value.as_str())
}

/// Read through the blank line that ends a message head; returns the head and any bytes already read past it.
pub(crate) fn __tok_http_read_head(stream: &mut impl std::io::Read) -> Result<(String, Vec<u8>), &'static str> {
    let mut data = Vec::new();
    let mut buffer = [0u8; 4096];
    loop {
        if let Some(end) = data.windows(4).position(|window| window == b"\r\n\r\n") {
            let head = String::from_utf8(data[..end].to_vec()).map_err(|_| "InvalidUtf8")?;
            return Ok((head, data[end + 4..].to_vec()));
        }
        if data.len() > __TOK_HTTP_HEAD_LIMIT { return Err("Other"); }
        let read = stream.read(&mut buffer).map_err(|_| "Other")?;
        if read == 0 { return Err("Other"); }
        data.extend_from_slice(&buffer[..read]);
    }
}

pub(crate) fn __tok_http_parse_headers<'a>(lines: impl Iterator<Item = &'a str>) -> Result<Vec<(String, String)>, &'static str> {
    let mut headers = Vec::new();
    for line in lines {
        let (name, value) = line.split_once(':').ok_or("Other")?;
        if name.is_empty() || name.contains(' ') { return Err("Other"); }
        __tok_http_add_header(&mut headers, name, value);
    }
    Ok(headers)
}

pub(crate) fn __tok_http_content_length(headers: &[(String, String)]) -> Result<Option<usize>, &'static str> {
    match __tok_http_header(headers, "content-length") {
        None => Ok(None),
        Some(text) => {
            let length = text.parse::<usize>().map_err(|_| "Other")?;
            if length > __TOK_HTTP_BODY_LIMIT { return Err("Other"); }
            Ok(Some(length))
        }
    }
}

/// Decode a complete `Transfer-Encoding: chunked` body.
pub(crate) fn __tok_http_dechunk(mut data: &[u8]) -> Result<Vec<u8>, &'static str> {
    let mut body = Vec::new();
    loop {
        let end = data.windows(2).position(|window| window == b"\r\n").ok_or("Other")?;
        let size_text = std::str::from_utf8(&data[..end]).map_err(|_| "Other")?;
        let size_text = size_text.split(';').next().unwrap_or("").trim();
        let size = usize::from_str_radix(size_text, 16).map_err(|_| "Other")?;
        data = &data[end + 2..];
        if size == 0 { return Ok(body); }
        if data.len() < size + 2 || body.len() + size > __TOK_HTTP_BODY_LIMIT { return Err("Other"); }
        body.extend_from_slice(&data[..size]);
        data = &data[size + 2..];
    }
}

pub(crate) fn __tok_http_read_request(stream: &mut std::net::TcpStream) -> Result<__TokHttpRequest, &'static str> {
    use std::io::Read;
    let (head, mut body) = __tok_http_read_head(stream)?;
    let mut lines = head.split("\r\n");
    let mut parts = lines.next().ok_or("Other")?.split(' ');
    let method = parts.next().filter(|method| !method.is_empty()).ok_or("Other")?.to_owned();
    let target = parts.next().filter(|target| target.starts_with('/')).ok_or("Other")?;
    if !parts.next().is_some_and(|version| version.starts_with("HTTP/1.")) { return Err("Other"); }
    let (path, query) = target.split_once('?').unwrap_or((target, ""));
    let headers = __tok_http_parse_headers(lines)?;
    if __tok_http_header(&headers, "transfer-encoding").is_some() { return Err("Other"); }
    let length = __tok_http_content_length(&headers)?.unwrap_or(0);
    while body.len() < length {
        let mut buffer = vec![0u8; (length - body.len()).min(64 * 1024)];
        let read = stream.read(&mut buffer).map_err(|_| "Other")?;
        if read == 0 { return Err("Other"); }
        body.extend_from_slice(&buffer[..read]);
    }
    body.truncate(length);
    let body = String::from_utf8(body).map_err(|_| "InvalidUtf8")?;
    Ok(__TokHttpRequest { method, path: path.to_owned(), query: query.to_owned(), headers, body })
}

pub(crate) fn __tok_http_write_response(stream: &mut std::net::TcpStream, response: &__TokHttpResponse) -> std::io::Result<()> {
    use std::io::Write;
    let status = if (100..=999).contains(&response.status) { response.status } else { 500 };
    let mut head = format!("HTTP/1.1 {status} {}\r\n", __tok_http_reason(status));
    for (name, value) in &response.headers {
        let name = name.to_ascii_lowercase();
        if name == "content-length" || name == "connection" || name.contains(['\r', '\n']) || value.contains(['\r', '\n']) { continue; }
        head.push_str(&format!("{name}: {value}\r\n"));
    }
    head.push_str(&format!("content-length: {}\r\nconnection: close\r\n\r\n", response.body.len()));
    stream.write_all(head.as_bytes())?;
    stream.write_all(response.body.as_bytes())?;
    stream.flush()
}

/// Answer `limit` requests one at a time (`limit <= 0` serves forever); the
/// handler returns `None` to stop early. Malformed requests get a 400 without
/// reaching the handler.
pub(crate) fn __tok_http_serve(addr: &str, limit: i32, handle: &mut dyn FnMut(__TokHttpRequest) -> Option<__TokHttpResponse>) -> Result<(), &'static str> {
    let listener = std::net::TcpListener::bind(addr).map_err(|_| "Other")?;
    let mut served = 0;
    for stream in listener.incoming() {
        let Ok(mut stream) = stream else { continue };
        let _ = stream.set_read_timeout(Some(std::time::Duration::from_secs(30)));
        let response = match __tok_http_read_request(&mut stream) {
            Ok(request) => match handle(request) { Some(response) => response, None => return Ok(()) },
            Err(_) => __TokHttpResponse { status: 400, headers: Vec::new(), body: "bad request".to_owned() },
        };
        let _ = __tok_http_write_response(&mut stream, &response);
        served += 1;
        if limit > 0 && served >= limit { return Ok(()); }
    }
    Ok(())
}

/// Split `http://host[:port][/path][?query]` into the connect address, Host header, and request target.
pub(crate) fn __tok_http_split_url(url: &str) -> Result<(String, String, String), &'static str> {
    let rest = url.strip_prefix("http://").ok_or("Other")?;
    let (authority, target) = match rest.find(['/', '?']) {
        Some(at) => (&rest[..at], &rest[at..]),
        None => (rest, "/"),
    };
    if authority.is_empty() || authority.contains('@') { return Err("Other"); }
    let target = if target.starts_with('?') { format!("/{target}") } else { target.to_owned() };
    let addr = if authority.rsplit_once(':').is_some_and(|(_, port)| !port.is_empty() && port.bytes().all(|b| b.is_ascii_digit())) {
        authority.to_owned()
    } else {
        format!("{authority}:80")
    };
    Ok((addr, authority.to_owned(), target))
}

/// HTTPS through the system `curl` (or `TOKIT_CURL`), which brings the
/// platform's TLS stack and certificate store. Redirects are not followed,
/// like plain HTTP; curl decodes chunked bodies itself.
pub(crate) fn __tok_https_request(grant: Option<&str>, method: &str, url: &str, headers: &[(String, String)], body: &str) -> Result<__TokHttpResponse, &'static str> {
    use std::io::Write;
    let rest = url.strip_prefix("https://").ok_or("Other")?;
    let authority = &rest[..rest.find(['/', '?']).unwrap_or(rest.len())];
    if authority.is_empty() || authority.contains('@') || url.contains(|c: char| c.is_whitespace()) { return Err("Other"); }
    let addr = if authority.rsplit_once(':').is_some_and(|(_, port)| !port.is_empty() && port.bytes().all(|b| b.is_ascii_digit())) {
        authority.to_owned()
    } else {
        format!("{authority}:443")
    };
    if !__tok_http_allowed(grant, &addr) { return Err("Denied"); }
    if method.is_empty() || method.contains([' ', '\r', '\n']) { return Err("Other"); }
    let curl = std::env::var_os("TOKIT_CURL").unwrap_or_else(|| "curl".into());
    let mut command = std::process::Command::new(curl);
    command.args(["--silent", "--show-error", "--include", "--http1.1", "--proto", "=https", "--max-time", "30", "--max-filesize", "16777216", "--request", method, "--header", "Expect:"]);
    for (name, value) in headers {
        let lower = name.to_ascii_lowercase();
        if matches!(lower.as_str(), "host" | "connection" | "content-length" | "expect") || name.contains(['\r', '\n', ':']) || value.contains(['\r', '\n']) { continue; }
        command.arg("--header").arg(format!("{lower}: {value}"));
    }
    if !body.is_empty() { command.args(["--data-binary", "@-"]); }
    let mut child = command.arg("--").arg(url)
        .stdin(std::process::Stdio::piped()).stdout(std::process::Stdio::piped()).stderr(std::process::Stdio::null())
        .spawn().map_err(|_| "Other")?;
    if let Some(mut stdin) = child.stdin.take() {
        stdin.write_all(body.as_bytes()).map_err(|_| "Other")?;
    }
    let output = child.wait_with_output().map_err(|_| "Other")?;
    if !output.status.success() { return Err("Other"); }
    let raw = output.stdout;
    let split = raw.windows(4).position(|window| window == b"\r\n\r\n").ok_or("Other")?;
    let head = std::str::from_utf8(&raw[..split]).map_err(|_| "Other")?;
    let mut lines = head.split("\r\n");
    let mut parts = lines.next().ok_or("Other")?.splitn(3, ' ');
    if !parts.next().is_some_and(|version| version.starts_with("HTTP/1.")) { return Err("Other"); }
    let status = parts.next().and_then(|code| code.parse::<i32>().ok()).ok_or("Other")?;
    let headers = __tok_http_parse_headers(lines)?;
    let body = String::from_utf8(raw[split + 4..].to_vec()).map_err(|_| "InvalidUtf8")?;
    Ok(__TokHttpResponse { status, headers, body })
}

/// Send one request and read the whole response; `http://` URLs go over a
/// plain socket and `https://` URLs through the system TLS client.
pub(crate) fn __tok_http_request(grant: Option<&str>, method: &str, url: &str, headers: &[(String, String)], body: &str) -> Result<__TokHttpResponse, &'static str> {
    use std::io::{Read, Write};
    if url.starts_with("https://") { return __tok_https_request(grant, method, url, headers, body); }
    let (addr, host, target) = __tok_http_split_url(url)?;
    if !__tok_http_allowed(grant, &addr) { return Err("Denied"); }
    if method.is_empty() || method.contains([' ', '\r', '\n']) || target.contains([' ', '\r', '\n']) { return Err("Other"); }
    let mut stream = std::net::TcpStream::connect(&addr).map_err(|_| "Other")?;
    let _ = stream.set_read_timeout(Some(std::time::Duration::from_secs(30)));
    let mut head = format!("{method} {target} HTTP/1.1\r\nhost: {host}\r\nconnection: close\r\ncontent-length: {}\r\n", body.len());
    for (name, value) in headers {
        let lower = name.to_ascii_lowercase();
        if matches!(lower.as_str(), "host" | "connection" | "content-length") || name.contains(['\r', '\n', ':']) || value.contains(['\r', '\n']) { continue; }
        head.push_str(&format!("{lower}: {value}\r\n"));
    }
    head.push_str("\r\n");
    stream.write_all(head.as_bytes()).map_err(|_| "Other")?;
    stream.write_all(body.as_bytes()).map_err(|_| "Other")?;
    let (head, mut rest) = __tok_http_read_head(&mut stream)?;
    let mut lines = head.split("\r\n");
    let status_line = lines.next().ok_or("Other")?;
    let mut parts = status_line.splitn(3, ' ');
    if !parts.next().is_some_and(|version| version.starts_with("HTTP/1.")) { return Err("Other"); }
    let status = parts.next().and_then(|code| code.parse::<i32>().ok()).ok_or("Other")?;
    let headers = __tok_http_parse_headers(lines)?;
    let mut limited = (&mut stream).take(__TOK_HTTP_BODY_LIMIT as u64 + 1);
    limited.read_to_end(&mut rest).map_err(|_| "Other")?;
    if rest.len() > __TOK_HTTP_BODY_LIMIT { return Err("Other"); }
    let body = if __tok_http_header(&headers, "transfer-encoding").is_some_and(|value| value.to_ascii_lowercase().contains("chunked")) {
        __tok_http_dechunk(&rest)?
    } else {
        match __tok_http_content_length(&headers)? {
            Some(length) if rest.len() >= length => { rest.truncate(length); rest }
            Some(_) => return Err("Other"),
            None => rest,
        }
    };
    let body = String::from_utf8(body).map_err(|_| "InvalidUtf8")?;
    Ok(__TokHttpResponse { status, headers, body })
}

// Raw TCP connections. Copies of a connection share one socket.
pub(crate) type __TokSocket = std::sync::Arc<std::sync::Mutex<Option<std::net::TcpStream>>>;

/// Connect to `addr` when the `--allow-net` grant covers it.
pub(crate) fn __tok_tcp_connect(grant: Option<&str>, addr: &str) -> Result<__TokSocket, &'static str> {
    if !__tok_http_allowed(grant, addr) { return Err("Denied"); }
    let stream = std::net::TcpStream::connect(addr).map_err(|_| "Other")?;
    let _ = stream.set_nodelay(true);
    Ok(std::sync::Arc::new(std::sync::Mutex::new(Some(stream))))
}

pub(crate) fn __tok_tcp_send(socket: &__TokSocket, data: &[u8]) -> Result<(), &'static str> {
    use std::io::Write;
    let mut guard = socket.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    let stream = guard.as_mut().ok_or("Other")?;
    stream.write_all(data).and_then(|()| stream.flush()).map_err(|_| "Other")
}

/// Up to `max` bytes (at least one unless the peer closed the connection).
pub(crate) fn __tok_tcp_recv(socket: &__TokSocket, max: i32) -> Result<Vec<u8>, &'static str> {
    use std::io::Read;
    let size = usize::try_from(max).ok().filter(|size| *size > 0).ok_or("Other")?.min(__TOK_HTTP_BODY_LIMIT);
    let mut guard = socket.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    let stream = guard.as_mut().ok_or("Other")?;
    let mut buffer = vec![0u8; size];
    let read = stream.read(&mut buffer).map_err(|_| "Other")?;
    buffer.truncate(read);
    Ok(buffer)
}

pub(crate) fn __tok_tcp_close(socket: &__TokSocket) {
    let mut guard = socket.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    if let Some(stream) = guard.take() {
        let _ = stream.shutdown(std::net::Shutdown::Both);
    }
}

/// Run `program` with `args` when the `--allow-run` grant names it (or is `*`),
/// feeding `stdin` and capturing up to 16 MiB of each output stream.
pub(crate) fn __tok_exec(grant: Option<&str>, program: &str, args: &[String], stdin: &str) -> Result<(i32, String, String), &'static str> {
    use std::io::{Read, Write};
    if !grant.is_some_and(|grant| grant == "*" || grant == program) { return Err("Denied"); }
    if program.is_empty() { return Err("NotFound"); }
    let mut child = std::process::Command::new(program).args(args)
        .stdin(std::process::Stdio::piped()).stdout(std::process::Stdio::piped()).stderr(std::process::Stdio::piped())
        .spawn().map_err(|error| if error.kind() == std::io::ErrorKind::NotFound { "NotFound" } else { "Other" })?;
    let input = stdin.as_bytes().to_vec();
    let mut pipe = child.stdin.take().ok_or("Other")?;
    let writer = std::thread::spawn(move || { let _ = pipe.write_all(&input); });
    let limit = __TOK_HTTP_BODY_LIMIT as u64;
    let mut out = child.stdout.take().ok_or("Other")?;
    let mut err = child.stderr.take().ok_or("Other")?;
    let errors = std::thread::spawn(move || { let mut buffer = Vec::new(); let _ = (&mut err).take(limit).read_to_end(&mut buffer); buffer });
    let mut stdout = Vec::new();
    (&mut out).take(limit).read_to_end(&mut stdout).map_err(|_| "Other")?;
    let stderr = errors.join().map_err(|_| "Other")?;
    let _ = writer.join();
    let status = child.wait().map_err(|_| "Other")?;
    let text = |bytes: Vec<u8>| String::from_utf8(bytes).map_err(|_| "InvalidUtf8");
    Ok((status.code().unwrap_or(-1), text(stdout)?, text(stderr)?))
}

// Listening sockets. Accepted connections are ordinary `Conn` sockets.
pub(crate) type __TokListener = std::sync::Arc<std::net::TcpListener>;

/// Bind `addr` when the `--allow-net` grant covers it.
pub(crate) fn __tok_tcp_listen(grant: Option<&str>, addr: &str) -> Result<__TokListener, &'static str> {
    if !__tok_http_allowed(grant, addr) { return Err("Denied"); }
    std::net::TcpListener::bind(addr).map(std::sync::Arc::new).map_err(|_| "Other")
}

/// Wait for the next connection.
pub(crate) fn __tok_tcp_accept(listener: &__TokListener) -> Result<__TokSocket, &'static str> {
    let (stream, _) = listener.accept().map_err(|_| "Other")?;
    let _ = stream.set_nodelay(true);
    Ok(std::sync::Arc::new(std::sync::Mutex::new(Some(stream))))
}

/// Read one HTTP/1.1 request from a connection, waiting at most 30 seconds.
pub(crate) fn __tok_http_read(socket: &__TokSocket) -> Result<__TokHttpRequest, &'static str> {
    let mut guard = socket.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    let stream = guard.as_mut().ok_or("Other")?;
    let _ = stream.set_read_timeout(Some(std::time::Duration::from_secs(30)));
    let request = __tok_http_read_request(stream);
    let _ = stream.set_read_timeout(None);
    request
}

/// Write one response with `content-length` and `connection: close`.
pub(crate) fn __tok_http_write(socket: &__TokSocket, response: &__TokHttpResponse) -> Result<(), &'static str> {
    let mut guard = socket.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    let stream = guard.as_mut().ok_or("Other")?;
    __tok_http_write_response(stream, response).map_err(|_| "Other")
}
