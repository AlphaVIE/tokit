#[derive(Clone, PartialEq)] enum __TokTaskError { Failed }
impl __TokRender for __TokTaskError {
    fn tok_render(&self) -> String { "TaskError::Failed".to_owned() }
}
struct __TokTaskState<T> {
    handle: Option<std::thread::JoinHandle<T>>,
    result: Option<Result<T, __TokTaskError>>,
}
#[derive(Clone)] struct __TokTask<T>(std::sync::Arc<std::sync::Mutex<__TokTaskState<T>>>);
impl<T> __TokRender for __TokTask<T> {
    fn tok_render(&self) -> String { "<task>".to_owned() }
}
#[cfg(not(target_family = "wasm"))]
const __TOK_MAX_ACTIVE_TASKS: usize = 64;
#[cfg(not(target_family = "wasm"))]
static __TOK_ACTIVE_TASKS: std::sync::atomic::AtomicUsize = std::sync::atomic::AtomicUsize::new(0);
#[cfg(not(target_family = "wasm"))]
struct __TokTaskSlot;
#[cfg(not(target_family = "wasm"))]
impl Drop for __TokTaskSlot {
    fn drop(&mut self) {
        __TOK_ACTIVE_TASKS.fetch_sub(1, std::sync::atomic::Ordering::AcqRel);
    }
}
#[cfg(not(target_family = "wasm"))]
fn __tok_task_from_spawn<T>(started: std::io::Result<std::thread::JoinHandle<T>>) -> __TokTask<T> {
    let (handle, result) = match started {
        Ok(handle) => (Some(handle), None),
        Err(_) => (None, Some(Err(__TokTaskError::Failed))),
    };
    __TokTask(std::sync::Arc::new(std::sync::Mutex::new(__TokTaskState { handle, result })))
}
#[cfg(not(target_family = "wasm"))]
fn __tok_spawn<T: Send + 'static>(job: impl FnOnce() -> T + Send + 'static) -> __TokTask<T> {
    __tok_spawn_limited(job, __TOK_MAX_ACTIVE_TASKS)
}
#[cfg(not(target_family = "wasm"))]
fn __tok_spawn_limited<T: Send + 'static>(job: impl FnOnce() -> T + Send + 'static, limit: usize) -> __TokTask<T> {
    use std::sync::atomic::Ordering;
    if __TOK_ACTIVE_TASKS.fetch_update(Ordering::AcqRel, Ordering::Acquire, |active| {
        (active < limit).then(|| active + 1)
    }).is_err() {
        return __tok_task_from_spawn(Err(std::io::Error::other("task capacity reached")));
    }
    let started = std::thread::Builder::new().stack_size(__TOK_STACK_BYTES).spawn(move || {
        let _slot = __TokTaskSlot;
        job()
    });
    if started.is_err() {
        __TOK_ACTIVE_TASKS.fetch_sub(1, Ordering::AcqRel);
    }
    __tok_task_from_spawn(started)
}
// Spawned functions are pure, so running them at the spawn site is a valid schedule.
#[cfg(target_family = "wasm")]
fn __tok_spawn<T: Send + 'static>(job: impl FnOnce() -> T + Send + 'static) -> __TokTask<T> {
    __TokTask(std::sync::Arc::new(std::sync::Mutex::new(__TokTaskState { handle: None, result: Some(Ok(job())) })))
}
fn __tok_join<T: Clone>(task: __TokTask<T>) -> Result<T, __TokTaskError> {
    let mut state = task.0.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    if state.result.is_none() {
        state.result = Some(match state.handle.take() {
            Some(handle) => handle.join().map_err(|_| __TokTaskError::Failed),
            None => Err(__TokTaskError::Failed),
        });
    }
    state.result.as_ref().expect("task result").clone()
}
