(~ http json task)
(# User (id i32) (name str))
(^ identity (T) ((x T)) T x)
(^ users_response () ((users [User])) (? Response Error)
  (= selected [User] [])
  (* user users
    (% (> (. user id) 0)
      (+ selected (identity user))))
  (= body str (! (json.encode selected)))
  (http.response 200 body))
(^ main () () (? Unit Error)
  (= server Server (http.server))
  (http.get server "GET" "/users" users_response)
  (= worker [Task (? Unit Error)] (task.spawn (^ () (http.run server))))
  (! (task.join worker)))
